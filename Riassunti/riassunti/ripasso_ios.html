1. Fondamenti di Swift

let / var

  • let crea una costante, var una variabile. Il tipo è dedotto dal compilatore (type inference) e si può indicare con : Tipo.
Swift
let pi = 3.14159          // costante
var eta: Int = 30         // variabile con tipo esplicito
eta += 1

if let / guard let

  • Servono per l'unwrapping sicuro di un Optional. if let esegue il blocco solo se il valore esiste, guard let esce subito dalla funzione se manca.
Swift
var nome: String? = "Mario"
if let nome {                     // qui nome è String
    print("Ciao \(nome)")
} else {
    print("Nessun nome")
}
func saluta(_ n: String?) {
    guard let n else { return }   // esce se nil
    print(n)
}

Optionals e operatore ??

  • Un Optional (Tipo?) contiene un valore oppure nil; è un enum con .none e .some. L'operatore ?? fornisce un valore di default se è nil.
Swift
let nickname: String? = nil
print(nickname ?? "ospite")   // "ospite"
let n = nickname?.count       // optional chaining -> Int?
// nickname!  -> crash se nil: da evitare

switch

  • Funziona con qualsiasi tipo e deve essere esaustivo (serve default). Ammette più valori per caso e la clausola where.
Swift
let v = "red pepper"
switch v {
case "celery": print("Uvetta")
case "cucumber", "watercress": print("Tramezzino")
case let x where x.hasSuffix("pepper"): print("Piccante: \(x)")
default: print("Zuppa")
}

for-in

  • Scorre array, dizionari e intervalli. 0..<4 esclude l'estremo superiore, 0...4 lo include.
Swift
for i in 0..<4 { print(i) }              // 0 1 2 3
for (chiave, valori) in ["a": [1, 2]] {  // dizionario
    print(chiave, valori)
}

while / repeat-while

  • while controlla la condizione prima, repeat-while dopo: il blocco gira almeno una volta.
Swift
var n = 2
while n < 100 { n *= 2 }          // 128
repeat { n *= 2 } while n < 100   // esegue almeno una volta

2. Strutture Dati e OOP

Classi e oggetti

  • La class è un tipo per riferimento: due variabili possono puntare allo stesso oggetto. Non serve new, si chiama direttamente l'inizializzatore.
Swift
class Shape {
    var name: String
    init(name: String) { self.name = name }   // costruttore
    func area() -> Double { 0 }
}
let s = Shape(name: "Forma")                  // niente 'new'

init

  • Imposta le proprietà iniziali e usa self per distinguerle dai parametri. deinit fa la pulizia prima della deallocazione.
Swift
class Persona {
    var nome: String
    var eta = 0                         // valore di default
    init(nome: String) { self.nome = nome }
    deinit { print("Persona rilasciata") }
}

Ereditarietà e override

  • Si eredita con : e ogni metodo ridefinito richiede override. Nell'init della sottoclasse prima si assegnano le proprietà proprie, poi si chiama super.init.
Swift
class Square: Shape {
    var side: Double
    init(side: Double) {
        self.side = side              // 1. proprietà proprie
        super.init(name: "Quadrato")  // 2. superclasse
    }
    override func area() -> Double { side * side }
}

struct

  • Tipo per valore: viene copiata a ogni assegnazione. Ha un init memberwise automatico e i metodi che modificano le proprietà sono mutating. Le view SwiftUI sono struct.
Swift
struct Punto {
    var x: Int
    var y: Int
    mutating func sposta(dx: Int) { x += dx }
}
var a = Punto(x: 1, y: 2)
var b = a        // copia
b.x = 99         // a.x resta 1

enum

  • Definisce un insieme chiuso di casi, anche con valori associati. CaseIterable fornisce allCases.
Swift
enum Pagamento {
    case contanti
    case carta(numero: String)      // valore associato
}
switch Pagamento.carta(numero: "1234") {
case .contanti: print("Contanti")
case .carta(let n): print(n)
}

Protocolli

  • Equivalente delle interface Java: elencano i requisiti che un tipo deve implementare. Con extension si aggiunge la conformità anche a tipi esistenti come Int.
Swift
protocol Descrivibile {
    var descrizione: String { get }
}
extension Int: Descrivibile {
    var descrizione: String { "Il numero \(self)" }
}
print(7.descrizione)

Closure

  • Un blocco di codice salvabile in una variabile o passabile come parametro. $0 è il primo argomento; la trailing closure si scrive fuori dalle parentesi.
Swift
let somma: (Int, Int) -> Int = { a, b in a + b }
let quadrati = [1, 2, 3, 4]
    .filter { $0 % 2 == 0 }   // [2, 4]
    .map { $0 * $0 }          // [4, 16]

3. Gestione Errori

throw

  • Un errore è un tipo (di solito un enum) che conforma a Error. La funzione che può fallire si marca con throws e segnala l'errore con throw.
Swift
enum ErroreLogin: Error { case passwordCorta, utenteVuoto }

func login(user: String, pwd: String) throws -> String {
    guard !user.isEmpty else { throw ErroreLogin.utenteVuoto }
    guard pwd.count >= 6 else { throw ErroreLogin.passwordCorta }
    return "Benvenuto \(user)"
}

do-catch

  • Si chiama la funzione con try dentro do e si gestisce ogni errore nei blocchi catch. L'ultimo catch generico riceve error in automatico.
Swift
do {
    print(try login(user: "Mario", pwd: "abc"))
} catch ErroreLogin.passwordCorta {
    print("Password corta")
} catch {
    print("Altro: \(error)")
}

try?

  • Trasforma l'errore in nil e il risultato in Optional, utile quando i dettagli dell'errore non servono. try! invece va in crash.
Swift
let esito = try? login(user: "Mario", pwd: "segreta")
print(esito ?? "Login fallito")

4. SwiftUI Basi

@main, App e WindowGroup

  • @main marca il punto d'ingresso; la struct conforma ad App e il suo body è una Scene. Il WindowGroup contiene la prima vista. Sostituisce l'AppDelegate usato prima di iOS 14.
Swift
@main
struct MiaApp: App {
    var body: some Scene {
        WindowGroup { ContentView() }
    }
}

some View

  • Ogni vista è una struct che conforma a View; il body restituisce some View (tipo opaco). Al posto dei cicli di vita UIKit si usano .onAppear e .onDisappear.
Swift
struct ContentView: View {
    var body: some View {
        Text("Ciao")
            .onAppear { print("Apparsa") }
    }
}

Modificatori base

  • Sono metodi che restituiscono una nuova vista, quindi l'ordine conta: .padding() prima di .background() colora anche il margine. .layoutPriority decide chi viene troncato per ultimo.
Swift
Text("Titolo")
    .font(.largeTitle)
    .padding()
    .background(Color.yellow)   // colora anche il padding
    .layoutPriority(1)          // priorità alta: si tronca per ultimo

VStack, HStack, ZStack

  • VStack impila in verticale, HStack in orizzontale, ZStack sovrappone le viste dal fondo verso l'alto. Accettano alignment e spacing.
Swift
VStack(alignment: .leading, spacing: 8) {
    HStack { Text("A"); Text("B") }
    ZStack {
        RoundedRectangle(cornerRadius: 12).fill(.blue)
        Text("Sopra").foregroundStyle(.white)
    }
    .frame(height: 80)
}

5. Gestione dello Stato

@State

  • Stato locale posseduto dalla vista: quando cambia, la UI si ridisegna. Va dichiarato private.
Swift
struct Contatore: View {
    @State private var n = 0
    var body: some View {
        Button("Tap: \(n)") { n += 1 }
    }
}

@Binding

  • Riferimento in lettura e scrittura a uno stato posseduto da un'altra vista. Il padre lo passa con il prefisso $.
Swift
struct Padre: View {
    @State private var acceso = false
    var body: some View { Figlio(isOn: $acceso) }   // passa il binding
}
struct Figlio: View {
    @Binding var isOn: Bool
    var body: some View { Button("Toggle") { isOn.toggle() } }
}

@Environment

  • Legge valori di sistema tramite key path: dark mode, size class, dismiss, modelContext.
Swift
struct Tema: View {
    @Environment(\.colorScheme) private var scheme
    var body: some View {
        Text(scheme == .dark ? "Dark" : "Light")
    }
}

@EnvironmentObject

  • Inietta un tuo oggetto condiviso a tutta la gerarchia. Deve conformare a ObservableObject e usare @Published per notificare i cambiamenti.
Swift
class Carrello: ObservableObject {
    @Published var articoli: [String] = []
}
struct Shop: View {
    @EnvironmentObject var carrello: Carrello
    var body: some View {
        Button("Aggiungi") { carrello.articoli.append("Pizza") }
    }
}
// Alla radice: Shop().environmentObject(Carrello())

Differenze pratiche

  • @State = dato mio. @Binding = dato del padre che posso modificare. @Environment = valore di sistema. @EnvironmentObject = oggetto mio condiviso ovunque.
Swift
@State private var testo = ""                 // lo possiedo io
@Binding var testo: String                    // appartiene al padre
@Environment(\.dismiss) var dismiss           // valore di sistema
@EnvironmentObject var carrello: Carrello     // oggetto condiviso

6. UI e Navigazione

Pulsanti personalizzati

  • Si creano con un ButtonStyle riutilizzabile o con una vista nell'etichetta del Button. configuration.isPressed permette di reagire alla pressione.
Swift
struct PrimaryStyle: ButtonStyle {
    func makeBody(configuration: Configuration) -> some View {
        configuration.label
            .padding()
            .background(Color.purple)
            .foregroundStyle(.white)
            .clipShape(RoundedRectangle(cornerRadius: 10))
            .opacity(configuration.isPressed ? 0.7 : 1)
    }
}
// Button("Salva") { }.buttonStyle(PrimaryStyle())

TextField e SecureField

  • Leggono e scrivono il testo tramite un binding ($var). SecureField nasconde i caratteri ed è pensato per le password.
Swift
@State private var user = ""
@State private var pwd = ""
// nel body:
TextField("Username", text: $user)
SecureField("Password", text: $pwd)

Toggle, Slider, Stepper

  • Tre controlli collegati a un binding: Toggle per i booleani, Slider per un intervallo continuo, Stepper per incrementi discreti.
Swift
@State private var on = true
@State private var vol = 50.0
@State private var q = 1
// nel body:
Toggle("Notifiche", isOn: $on)
Slider(value: $vol, in: 0...100)
Stepper("Quantità: \(q)", value: $q, in: 1...10)

Picker

  • Offre una scelta tra opzioni e salva la selezione in un binding. Stili: menu (default), .wheel, .segmented. Dentro un Form richiede una NavigationStack.
Swift
@State private var tema = 0
Picker("Tema", selection: $tema) {
    Text("Chiaro").tag(0)
    Text("Scuro").tag(1)
}
.pickerStyle(.segmented)

Form

  • Contenitore che raggruppa i controlli di input in Section, con stile da impostazioni e scroll integrato.
Swift
NavigationStack {
    Form {
        Section("Account") {
            TextField("Nome", text: $user)
        }
    }
    .navigationTitle("Impostazioni")
}

List con ForEach e .onDelete

  • List mostra righe scorrevoli. .onDelete e .onMove funzionano solo su ForEach (conforme a DynamicViewContent). Per il riordino serve EditButton.
Swift
@State private var cose = ["Studiare", "Spesa", "Cucinare"]
List {
    ForEach(cose, id: \.self) { Text($0) }
        .onDelete { cose.remove(atOffsets: $0) }
        .onMove { cose.move(fromOffsets: $0, toOffset: $1) }
}
.toolbar { EditButton() }

TabView

  • Crea la barra a schede in basso. Ogni scheda è una vista con il modificatore .tabItem.
Swift
TabView {
    Text("Home").tabItem { Label("Home", systemImage: "house.fill") }
    Text("Profilo").tabItem { Label("Profilo", systemImage: "person") }
}

ScrollView

  • Rende scorrevole qualsiasi contenuto. Con ScrollViewReader si scorre in modo programmatico a una vista con un certo id.
Swift
ScrollViewReader { proxy in
    ScrollView {
        Button("Vai in fondo") { proxy.scrollTo(99) }
        ForEach(0..<100, id: \.self) { Text("Riga \($0)").id($0) }
    }
}

NavigationStack e NavigationLink

  • NavigationStack gestisce la navigazione a pila, NavigationLink apre la schermata di destinazione. In una List il tipo Identifiable evita di passare id:.
Swift
struct Ristorante: Identifiable {
    let id = UUID()
    let nome: String
}
NavigationStack {
    List(lista) { r in
        NavigationLink(r.nome) { Text("Dettaglio \(r.nome)") }
    }
    .navigationTitle("Ristoranti")
}

7. Servizi Web e Mappe

URLSession, Codable, Identifiable

  • Il modello conforma a Codable (JSON ↔ oggetto) e Identifiable (id univoco per le liste). URLSession è asincrona: oggi si usa async/await con .task.
Swift
struct Book: Codable, Identifiable {
    let id: Int
    let title: String
}
@MainActor
class BookService: ObservableObject {
    @Published var books: [Book] = []
    func load() async {
        guard let url = URL(string: "https://www.mysite.com/books.json") else { return }
        do {
            let (data, _) = try await URLSession.shared.data(from: url)
            books = try JSONDecoder().decode([Book].self, from: data)
        } catch { print(error) }
    }
}
// Nella vista: List(service.books) { Text($0.title) }.task { await service.load() }

AsyncImage

  • Scarica e mostra un'immagine da URL, con un placeholder durante il caricamento. La versione con phase gestisce anche l'errore.
Swift
AsyncImage(url: URL(string: "https://picsum.photos/400")) { phase in
    switch phase {
    case .success(let img): img.resizable().scaledToFit()
    case .failure: Image(systemName: "photo.badge.exclamationmark")
    default: ProgressView()          // in caricamento
    }
}
.frame(width: 300, height: 300)

Mappe interattive con MapKit

  • Map mostra un'area interattiva, marker e posizione utente. Serve la chiave NSLocationWhenInUseUsageDescription nell'Info.plist. Map(coordinateRegion:) è deprecato da iOS 17: qui la versione moderna.
Swift
import MapKit

struct MappaView: View {
    @State private var camera: MapCameraPosition = .region(
        MKCoordinateRegion(
            center: .init(latitude: 45.4642, longitude: 9.19),
            span: .init(latitudeDelta: 0.05, longitudeDelta: 0.05)))
    var body: some View {
        Map(position: $camera) {
            UserAnnotation()                      // posizione utente
            Marker("Duomo", systemImage: "star.fill",
                   coordinate: .init(latitude: 45.4641, longitude: 9.1919))
        }
        .mapControls { MapUserLocationButton() }
    }
}

8. Persistenza

UserDefaults e @AppStorage

  • UserDefaults salva piccole preferenze in un plist caricato all'avvio: oltre circa 512 KB l'app rallenta. @AppStorage è il wrapper SwiftUI che salva e aggiorna la UI da solo.
Swift
struct TapView: View {
    @AppStorage("tapCount") private var tapCount = 0   // chiave + default
    var body: some View {
        Button("Tap: \(tapCount)") { tapCount += 1 }   // persiste da solo
    }
}
// Basso livello: UserDefaults.standard.set(5, forKey: "tapCount")

SwiftData e @Model

  • Framework dichiarativo introdotto con iOS 17. @Model rende una classe persistente; le relazioni si esprimono con proprietà come [Person].
Swift
import SwiftData

@Model
class Person {
    var name: String
    var age: Int
    init(name: String, age: Int) { self.name = name; self.age = age }
}
@Model
class Company {
    var name: String
    var employees: [Person] = []     // relazione uno-a-molti
    init(name: String) { self.name = name }
}

.modelContainer

  • Configura il database all'avvio e dichiara quali modelli gestire. Si applica alla scena o alla vista radice.
Swift
@main
struct PersoneApp: App {
    var body: some Scene {
        WindowGroup { PersoneView() }
            .modelContainer(for: Person.self)   // crea il database
    }
}

@Query e .insert()

  • @Query legge i dati e aggiorna la vista a ogni modifica. Per scrivere si prende il contesto da @Environment(\.modelContext) e si chiama insert(); save() è facoltativo.
Swift
struct PersoneView: View {
    @Environment(\.modelContext) private var context
    @Query(sort: \Person.name) private var people: [Person]   // lista reattiva

    var body: some View {
        List(people) { Text("\($0.name), \($0.age) anni") }
        Button("Aggiungi") {
            context.insert(Person(name: "Mario", age: 30))    // inserimento
            try? context.save()
        }
    }
}
